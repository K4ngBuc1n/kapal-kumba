Panduan Deployment Google Apps Script

Ikuti langkah berikut untuk mengintegrasikan dashboard ini dengan Google Sheets sebagai database utama melalui Google Apps Script (GAS):

  1. Buka Google Sheets baru, beri nama database (misal: Database_Maintenance_Engine).
  2. Klik menu Extensions / Ekstensi > Apps Script.
  3. Salin kode backend di bawah ke dalam file Code.gs.
  4. Buat file HTML baru di proyek Apps Script, beri nama index (akan menjadi index.html), lalu salin seluruh isi file dashboard ini ke dalamnya.
  5. Klik tombol Deploy / Terapkan > Deployment Baru (New Deployment).
  6. Pilih Jenis: Web App (Aplikasi Web). Akses (Who has access): Anyone (Siapa Saja). Klik Deploy dan berikan izin (*Authorize*).

function doGet() {
  return HtmlService.createHtmlOutputFromFile('index')
      .setTitle('Monitoring Maintenance Main Engine')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1.0');
}

// Simpan Data Baru dari Dashboard Form ke Spreadsheet
function saveDataToSheet(data) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "Data_Monitoring";
    var sheet = ss.getSheetByName(sheetName);
    
    // Jika Sheet Belum Ada, Buat Otomatis Beserta Header Kolom
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow([
        "Timestamp", 
        "Nama Komponen", 
        "Tingkat Defect", 
        "MTTR (Jam)", 
        "MTTF (Jam)", 
        "RUL (Jam)", 
        "Tindakan", 
        "Catatan"
      ]);
      sheet.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#0f172a").setFontColor("#ffffff");
    }
    
    sheet.appendRow([
      new Date(),
      data.componentName,
      data.defectLevel,
      parseFloat(data.mttr),
      parseFloat(data.mttf),
      parseFloat(data.rul),
      data.actionOption,
      data.notes || "-"
    ]);
    
    return { success: true, message: "Data berhasil disimpan ke Google Sheets!" };
  } catch (error) {
    return { success: false, message: "Terjadi Kesalahan: " + error.toString() };
  }
}

// Ambil Data dari Spreadsheet untuk Tampilan Dashboard
function getMonitoringData() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Data_Monitoring");
    if (!sheet) return [];
    
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return []; // Hanya header
    
    var results = [];
    for (var i = 1; i < data.length; i++) {
      results.push({
        id: i,
        timestamp: data[i][0],
        componentName: data[i][1],
        defectLevel: data[i][2],
        mttr: data[i][3],
        mttf: data[i][4],
        rul: data[i][5],
        actionOption: data[i][6],
        notes: data[i][7]
      });
    }
    return results;
  } catch (e) {
    return [];
  }
}
